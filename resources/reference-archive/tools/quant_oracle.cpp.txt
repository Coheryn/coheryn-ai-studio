// Development-only reference fixture generator. Never built or linked by Cargo.
#include "ggml.h"
#include "ggml-cpu.h"
#include <algorithm>
#include <cmath>
#include <cstdint>
#include <iomanip>
#include <iostream>
#include <vector>

static void floats(const std::vector<float>& values) {
    std::cout << '[';
    for (size_t i=0;i<values.size();++i) { if(i)std::cout<<','; std::cout<<std::setprecision(9)<<values[i]; }
    std::cout<<']';
}
static void bytes(const std::vector<uint8_t>& values) {
    std::cout<<'[';
    for(size_t i=0;i<values.size();++i) {if(i)std::cout<<',';std::cout<<unsigned(values[i]);}
    std::cout<<']';
}
int main() {
    ggml_cpu_init();
    std::cout << "{\"schema_version\":1,\"tolerances\":{\"decode_absolute\":0,\"activation_bytes_exact\":true,\"dot_absolute\":0.00001,\"dot_relative\":0.00001},\"cases\":[";
    bool comma=false;
    for(auto type:{GGML_TYPE_Q8_0,GGML_TYPE_Q4_0,GGML_TYPE_Q5_0,GGML_TYPE_Q4_K,GGML_TYPE_Q6_K}) {
        const auto * weights=ggml_get_type_traits(type);
        const auto * cpu=ggml_get_type_traits_cpu(type);
        const auto * activation=ggml_get_type_traits_cpu(cpu->vec_dot_type);
        const int block=int(ggml_blck_size(type));
        for(int blocks:{1,2,3,8}) for(int pattern=0;pattern<5;++pattern) {
            const int n=block*blocks;
            std::vector<float> w(n),x(n),decoded(n);
            for(int j=0;j<n;++j) {
                w[j]=float(((j*37+pattern*11)%251)-125)/32.f;
                x[j]=float(((j*19+pattern*7)%127)-63)/16.f;
                if(pattern==0) {w[j]=0;x[j]=0;}
                if(pattern==2) {w[j]=(j%2?-1.f:1.f)*float(j%29)/8.f; x[j]=(j%2?-1.f:1.f)*float(j%17)/4.f;}
                if(pattern==3) {w[j]=float(j%23)/8.f; x[j]=float(j%31)/16.f;}
                if(pattern==4) {w[j]=std::ldexp(w[j],-12); x[j]=std::ldexp(x[j],-8);}
            }
            // Value ties deliberately cross sign/scale and block boundaries.
            if(pattern==1) {x[0]=127.f;x[1]=0.5f;x[2]=-0.5f;x[3]=1.5f;x[4]=-1.5f;}
            std::vector<uint8_t> packed(ggml_row_size(type,n));
            std::vector<uint8_t> operand(ggml_row_size(cpu->vec_dot_type,n),0);
            weights->from_float_ref(w.data(),packed.data(),n);
            activation->from_float(x.data(),operand.data(),n);
            weights->to_float(packed.data(),decoded.data(),n);
            float dot=0;
            cpu->vec_dot(n,&dot,0,packed.data(),0,operand.data(),0,1);
            if(comma)std::cout<<',';comma=true;
            std::cout<<"{\"dtype\":"<<int(type)<<",\"blocks\":"<<blocks<<",\"pattern\":"<<pattern<<",\"bytes\":";
            bytes(packed);std::cout<<",\"activation_dtype\":"<<int(cpu->vec_dot_type)<<",\"activation_bytes\":";bytes(operand);
            std::cout<<",\"input\":";floats(x);std::cout<<",\"decoded\":";floats(decoded);
            std::cout<<",\"dot\":"<<std::setprecision(9)<<dot<<'}';
        }
    }
    std::cout<<"]}\n";
}
